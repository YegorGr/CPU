; this is an example of a program to solve quadratic equations
; if given equation has an infinite amount of roots it should return 666 if no roots return 0

IN
POPR AX
IN
POPR BX
IN
POPR CX

PUSHR AX
PUSH 0
JE linear ; if this is a linear equation

; if no a linear equation
; solve the discriment D = SQRT(bx*bx - 4*ax*cx)

PUSHR BX
PUSHR BX 
MUL
PUSH 4
PUSHR AX
MUL
PUSHR CX
MUL
SUB
POPR DDX

; if D < 0 
PUSHR DDX
PUSH 0
JB no_roots

; if D == 0
PUSHR DDX 
PUSH 0
JE one_root

; if D > 0
PUSHR DDX 
PUSH 0
JA two_roots

linear:
PUSHR CX
PUSH 0
JE inf_roots
PUSHR CX
PUSH -1
MUL
PUSHR BX
DIV
OUT 
HLT

no_roots:
PUSH 0
OUT
HLT

one_root:
PUSHR BX
PUSH -1
MUL
PUSH 2
DIV
PUSHR AX
DIV
OUT
HLT

two_roots:
PUSHR BX
PUSH -1
MUL
PUSHR DDX
SQRT
PUSH -1
MUL
SUB
PUSH 2
DIV
PUSHR AX
DIV
OUT

PUSHR BX
PUSH -1
MUL
PUSHR DDX
SUB
PUSH 2
DIV
PUSHR AX
DIV
OUT

HLT

inf_roots:
PUSH 666
OUT
HLT
